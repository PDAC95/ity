    **Reset-password page:**

    1. **Add session guard on mount:** Add state variables and useEffect to check for a valid session before rendering the form. Import `useEffect` from React.
    ```typescript
    const [sessionChecked, setSessionChecked] = useState(false);
    const [hasSession, setHasSession] = useState(false);

    useEffect(() => {
      const checkSession = async () => {
        const { data: { user } } = await supabase.auth.getUser();
        setHasSession(!!user);
        setSessionChecked(true);
        if (!user) {
          router.replace('/login?error=no_recovery_session');
        }
      };
      checkSession();
    }, [supabase, router]);

    if (!sessionChecked) return null; // Loading — renders nothing briefly
    if (!hasSession) return null; // Redirecting to login
    ```
    Place this BEFORE the form JSX return. This ensures direct navigation to /reset-password without a recovery session (from /auth/confirm) gets bounced to login.

    window.location.href = '/login?message=password_reset';
