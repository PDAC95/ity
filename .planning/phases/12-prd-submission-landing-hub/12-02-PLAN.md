**triggerPrdGeneration function:**
```typescript
async function triggerPrdGeneration() {
  setPrdFlow({ phase: 'generating' });
  try {
    const chatHistory = messages
      .filter(m => m.role === 'user' || m.role === 'assistant')
      .map(m => ({
        role: m.role,
        content: m.parts.filter((p): p is {type:'text';text:string} => p.type === 'text').map(p => p.text).join(''),
      }));

    const res = await fetch('/api/prd/generate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ chatHistory, schoolId, templateId }),
    });

    if (!res.ok) throw new Error('generation_failed');
    const { prdData } = await res.json();
    setPrdFlow({ phase: 'summary', prdData });
  } catch {
    // 1 automatic retry (transparent to user — CONTEXT.md)
    try {
      // Retry once more
      const chatHistory = messages
        .filter(m => m.role === 'user' || m.role === 'assistant')
        .map(m => ({
          role: m.role,
          content: m.parts.filter((p): p is {type:'text';text:string} => p.type === 'text').map(p => p.text).join(''),
        }));
      const res = await fetch('/api/prd/generate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ chatHistory, schoolId, templateId }),
      });
      if (!res.ok) throw new Error('retry_failed');
      const { prdData } = await res.json();
      setPrdFlow({ phase: 'summary', prdData });
    } catch {
      setPrdFlow({ phase: 'error', retryCount: 0 });
    }
  }
}
```

**Confirm handler:**
```typescript
const requestPageMutation = trpc.landing.requestPage.useMutation({
  onSuccess: () => setPrdFlow({ phase: 'done' }),
  onError: () => toast.error('Error al enviar la solicitud.'),
});

function handleConfirm() {
  if (prdFlow.phase !== 'summary') return;
  setPrdFlow({ phase: 'confirming', prdData: prdFlow.prdData });
  requestPageMutation.mutate({ schoolId, prdData: prdFlow.prdData as Record<string, unknown> });
}
```

**Change handler:**

**Manual retry handler:**
```typescript
function handleManualRetry() {
  if (prdFlow.phase !== 'error') return;
  const newRetryCount = prdFlow.retryCount + 1;
  if (newRetryCount > MAX_MANUAL_RETRIES) return;
  triggerPrdGeneration(); // will set phase to generating, then summary or error
}
```
Note: When error phase enters after manual retry attempt, increment retryCount. Adjust triggerPrdGeneration to accept optional retryCount parameter and set it on error.
